// auto-generated by Goal Loop — gpt-oss:20b · 1 iteration

export function chunk<T>(array: T[], size: number): T[][] {
  if (size <= 0) throw new Error('Chunk size must be greater than 0')
  const result: T[][] = []
  for (let i = 0; i < array.length; i += size)
    result.push(array.slice(i, i + size))
  return result
}

export function unique<T>(array: T[]): T[] {
  return Array.from(new Set(array))
}

export function flatten<T>(array: T[][]): T[] {
  return array.reduce((acc, val) => acc.concat(val), [] as T[])
}

export function groupBy<T>(
  array: T[],
  key: (item: T) => string
): Record<string, T[]> {
  return array.reduce((groups, item) => {
    const k = key(item)
    return { ...groups, [k]: [...(groups[k] ?? []), item] }
  }, {} as Record<string, T[]>)
}

// src/retry.ts
export async function retry<T>(
  fn: () => Promise<T>,
  maxAttempts: number,
  delayMs: number,
  options?: { signal?: AbortSignal; onRetry?: (attempt: number, error: unknown) => void }
): Promise<T> {
  const { signal, onRetry } = options ?? {};
  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    if (signal?.aborted) throw new Error('Aborted');
    try { return await fn(); }
    catch (err) {
      if (attempt === maxAttempts) throw err;
      onRetry?.(attempt, err);
      await pause(delayMs * Math.pow(2, attempt - 1), signal);
    }
  }
  throw new Error('Unexpected');
}
